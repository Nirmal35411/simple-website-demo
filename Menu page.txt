        function toggleCart() {
            const cartModal = document.getElementById('cart-modal');
            cartModal.classList.toggle('show');
        }

        // Category filtering
        function filterMenu(category) {
            const menuItems = document.querySelectorAll('.menu-item');
            const tabs = document.querySelectorAll('.category-tab');

            // Update active tab
            tabs.forEach(tab => {
                tab.classList.remove('active');
                if (tab.dataset.category === category) {
                    tab.classList.add('active');
                }
            });

            // Filter items
            menuItems.forEach(item => {
                if (category === 'all' || item.dataset.category === category) {
                    item.classList.remove('hidden');
                } else {
                    item.classList.add('hidden');
                }
            });
        }

        async function onConfigChange(config) {
            // Update text content
            const pageTitle = document.getElementById('page-title');
            if (pageTitle) pageTitle.textContent = config.page_title || defaultConfig.page_title;
            
            const cartText = document.getElementById('cart-text');
            if (cartText) cartText.textContent = config.cart_text || defaultConfig.cart_text;
            
            const viewCartText = document.getElementById('view-cart-text');
            if (viewCartText) viewCartText.textContent = config.view_cart_text || defaultConfig.view_cart_text;
            
            const emptyCartText = document.getElementById('empty-cart-text');
            if (emptyCartText) emptyCartText.textContent = config.empty_cart_text || defaultConfig.empty_cart_text;

            // Update all "Add to Cart" buttons
            const addToCartButtons = document.querySelectorAll('#add-to-cart-text');
            addToCartButtons.forEach(btn => {
                btn.textContent = config.add_to_cart_text || defaultConfig.add_to_cart_text;
            });

            // Update colors
            const backgroundColor = config.background_color || defaultConfig.background_color;
            const primaryColor = config.primary_color || defaultConfig.primary_color;
            const accentColor = config.accent_color || defaultConfig.accent_color;
            const textColor = config.text_color || defaultConfig.text_color;
            const surfaceColor = config.surface_color || defaultConfig.surface_color;

            document.body.style.backgroundColor = backgroundColor;
            document.body.style.color = textColor;

            // Update primary color elements
            const header = document.querySelector('.header');
            if (header) header.style.background = `linear-gradient(135deg, ${primaryColor} 0%, ${primaryColor}dd 100%)`;

            const addToCartBtns = document.querySelectorAll('.add-to-cart-btn');
            addToCartBtns.forEach(btn => {
                btn.style.background = `linear-gradient(135deg, ${primaryColor} 0%, ${primaryColor}dd 100%)`;
            });

            // Update accent color elements
            const floatingCart = document.querySelector('.floating-cart');
            if (floatingCart) floatingCart.style.backgroundColor = accentColor;

            const activeTab = document.querySelector('.category-tab.active');
            if (activeTab) {
                activeTab.style.backgroundColor = accentColor;
                activeTab.style.borderColor = accentColor;
            }

            const checkoutBtn = document.querySelector('.checkout-btn');
            if (checkoutBtn) checkoutBtn.style.backgroundColor = accentColor;

            // Update surface color elements
            const surfaceElements = document.querySelectorAll('.menu-item, .cart-content');
            surfaceElements.forEach(el => {
                el.style.backgroundColor = surfaceColor;
            });

            // Update font
            const customFont = config.font_family || defaultConfig.font_family;
            const titleElements = document.querySelectorAll('.page-title, .menu-title, .cart-title');
            titleElements.forEach(el => {
                el.style.fontFamily = `${customFont}, serif`;
            });

            // Update font size
            const baseSize = config.font_size || defaultConfig.font_size;
            document.documentElement.style.fontSize = `${baseSize}px`;
        }
