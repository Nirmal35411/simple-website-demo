  <script>
        // Configuration object for Element SDK
        const defaultConfig = {
            partner_name: "Alex Kumar",
            earnings_label: "Today's Earnings",
            online_status_text: "Online - Ready for deliveries",
            offline_status_text: "Offline - Not accepting orders"
        };

        let isOnline = true;
        let currentOrderId = null;

        // Element SDK initialization
        if (window.elementSdk) {
            window.elementSdk.init({
                defaultConfig: defaultConfig,
                onConfigChange: async (config) => {
                    // Update partner name
                    const partnerNameEl = document.getElementById('partner-name');
                    if (partnerNameEl) {
                        partnerNameEl.textContent = config.partner_name || defaultConfig.partner_name;
                    }

                    // Update earnings label
                    const earningsLabelEl = document.getElementById('earnings-title');
                    if (earningsLabelEl) {
                        earningsLabelEl.textContent = config.earnings_label || defaultConfig.earnings_label;
                    }

                    // Update status text based on current online state
                    const statusTextEl = document.getElementById('status-text');
                    if (statusTextEl) {
                        if (isOnline) {
                            statusTextEl.textContent = config.online_status_text || defaultConfig.online_status_text;
                        } else {
                            statusTextEl.textContent = config.offline_status_text || defaultConfig.offline_status_text;
                        }
                    }
                },
                mapToCapabilities: (config) => ({
                    recolorables: [],
                    borderables: [],
                    fontEditable: undefined,
                    fontSizeable: undefined
                }),
                mapToEditPanelValues: (config) => new Map([
                    ["partner_name", config.partner_name || defaultConfig.partner_name],
                    ["earnings_label", config.earnings_label || defaultConfig.earnings_label],
                    ["online_status_text", config.online_status_text || defaultConfig.online_status_text],
                    ["offline_status_text", config.offline_status_text || defaultConfig.offline_status_text]
                ])
            });
        }

        // Toggle availability function
        function toggleAvailability() {
            isOnline = !isOnline;
            const toggle = document.getElementById('availability-toggle');
            const statusText = document.getElementById('status-text');
            
            if (isOnline) {
                toggle.classList.add('active');
                toggle.setAttribute('aria-checked', 'true');
                statusText.className = 'status-text status-online';
                const config = window.elementSdk ? window.elementSdk.config : defaultConfig;
                statusText.textContent = config.online_status_text || defaultConfig.online_status_text;
            } else {
                toggle.classList.remove('active');
                toggle.setAttribute('aria-checked', 'false');
                statusText.className = 'status-text status-offline';
                const config = window.elementSdk ? window.elementSdk.config : defaultConfig;
                statusText.textContent = config.offline_status_text || defaultConfig.offline_status_text;
            }
        }

        // Handle keyboard navigation for toggle
        function handleToggleKeydown(event) {
            if (event.key === 'Enter' || event.key === ' ') {
                event.preventDefault();
                toggleAvailability();
            }
        }

        // Accept delivery function
        function acceptDelivery(deliveryId) {
            // Show loading state
            const button = event.target;
            const originalText = button.textContent;
            button.textContent = 'Accepting...';
            button.disabled = true;
            
            // Simulate API call
            setTimeout(() => {
                // Move delivery to active section
                moveToActiveDeliveries(deliveryId);
                
                // Update counters
                updateDeliveryCounters();
                
                // Show success message
                showToast('Delivery accepted successfully!');
                
                button.textContent = originalText;
                button.disabled = false;
            }, 1000);
        }

        // Move delivery to active section
        function moveToActiveDeliveries(deliveryId) {
            // This would typically involve API calls and state management
            // For demo purposes, we'll just update the UI
            const availableSection = document.getElementById('available-deliveries');
            const activeSection = document.getElementById('active-deliveries');
            
            // Remove from available (simplified for demo)
            const cards = availableSection.querySelectorAll('.delivery-card');
            if (cards[deliveryId - 1]) {
                cards[deliveryId - 1].style.display = 'none';
            }
        }

        // Update delivery counters
        function updateDeliveryCounters() {
            const availableCount = document.getElementById('available-count');
            const activeCount = document.getElementById('active-count');
            
            if (availableCount) {
                const currentCount = parseInt(availableCount.textContent);
                availableCount.textContent = Math.max(0, currentCount - 1);
            }
            
            if (activeCount) {
                const currentCount = parseInt(activeCount.textContent);
                activeCount.textContent = currentCount + 1;
            }
        }

        // Call customer function
        function callCustomer(phoneNumber) {
            // Create a temporary link to initiate call
            const link = document.createElement('a');
            link.href = `tel:${phoneNumber}`;
            link.click();
            
            showToast('Initiating call...');
        }

        // Open navigation function
        function openNavigation(address) {

        // Trap focus in modal
        function trapFocus(element) {
            const focusableElements = element.querySelectorAll(
                'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
            );
            const firstElement = focusableElements[0];
            const lastElement = focusableElements[focusableElements.length - 1];
            
            element.addEventListener('keydown', (e) => {
                if (e.key === 'Tab') {
                    if (e.shiftKey) {
                        if (document.activeElement === firstElement) {
                            lastElement.focus();
                            e.preventDefault();
                        }
                    } else {
                        if (document.activeElement === lastElement) {
                            firstElement.focus();
                            e.preventDefault();
                        }
                    }
                }
                
                if (e.key === 'Escape') {
                    closeOTPModal();
                }
            });
        }
